১. Google Sheet খুলে Extensions → Apps Script-এ নিচের কোডটি পেস্ট করে Deploy → New Deployment → Web App (Who has access: Anyone) করুন।

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var body = JSON.parse(e.postData.contents || '{}');
  var sheetName = body.sheet || 'Transactions';
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(['SyncTime', 'ID', 'JSON_Data']);
  }
  sheet.appendRow([new Date(), body.id || '', JSON.stringify(body.payload)]);
  return ContentService.createTextOutput(JSON.stringify({status: 'success'})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheets = ['Members', 'Ledger', 'Charmonai', 'Mahfil'];
  var result = { status: 'success' };
  sheets.forEach(function(name) {
    var sh = ss.getSheetByName(name);
    result[name] = sh ? sh.getDataRange().getValues() : [];
  });
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}
